1.CREATE TABLE FUNDRAISING_TEAM (
    F_ID INT PRIMARY KEY,
    F_NAME VARCHAR(300),
    F_DETAILS VARCHAR2(300)
);
DESC FUNDRAISING_TEAM;


2.CREATE TABLE EVENTS (
    E_ID INT PRIMARY KEY,
    E_LOCATION VARCHAR(260),
    E_TITLE VARCHAR(260),
    E_DATE DATE,
    E_TYPE VARCHAR(260),
    F_ID INT,
    FOREIGN KEY (F_ID) REFERENCES FUNDRAISING_TEAM(F_ID)
);
DESC EVENTS;

3.CREATE TABLE DONATION (
    D_AMOUNT DECIMAL(10,2),
    D_DATE DATE,
    E_ID INT,
    D_ID INT,
    ORG_ID INT,
    PRIMARY KEY (E_ID, D_ID),
    FOREIGN KEY (E_ID) REFERENCES EVENTS(E_ID),
    FOREIGN KEY (F_ID) REFERENCES FUNDRAISING_TEAM(F_ID)
);
DESC DONATION;


4.CREATE TABLE DONERS (
    DONER_ID INT PRIMARY KEY,
    DONER_PHONE VARCHAR(20),
    DONER_NAME VARCHAR(100),
    DONER_EMAIL VARCHAR(255),
ORG_ID NUMBER
);
DESC DONERS;

5.CREATE TABLE DONERSDONATION (
    D_ID INT PRIMARY KEY,
    D_AMOUNT DECIMAL(10,2),
    D_DATE DATE,
    ORG_ID INT,
    DONER_ID INT,
    FOREIGN KEY (DONER_ID) REFERENCES DONERS(DONER_ID)
);
DESC DONERSDONATION;

6.CREATE TABLE ORGANIZATION (
    ORG_ID INT PRIMARY KEY,
    ORG_CONTACT VARCHAR(20),
    ORG_NAME VARCHAR(100),
    ORG_ADDRESS VARCHAR(255)
);
DESC ORGANIZATION;
